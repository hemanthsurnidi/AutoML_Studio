How to Use Your Exported Model in Python

import joblib
import pandas as pd

# Load the bundle
bundle = joblib.load("automl_model.pkl")

model = bundle["model"]
ct = bundle["column_transformer"]
feature_names = bundle["feature_names"]
label_encoder = bundle.get("label_encoder")  # Classification only

# Prepare your new data (must have the same columns as training)
new_data = pd.DataFrame([{
{% for f in [] %}    "{{ f }}": ...,
{% endfor %}    # ... your feature values here
}])

# Apply the same preprocessing
X = ct.transform(new_data)

# Predict
prediction = model.predict(X)[0]

# Decode label for classification
if label_encoder:
    prediction = label_encoder.inverse_transform([int(prediction)])[0]

print("Prediction:", prediction)
